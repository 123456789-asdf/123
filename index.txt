function renderGallery() {
  const grid = $("#galleryGrid");
  if (!grid) return;
  const list = (resume.gallery || []).filter(function (g) { return g && g.src; });
  if (!list.length) {
    const sec = $("#gallery");
    if (sec) sec.style.display = "none";
    return;
  }
  grid.innerHTML = list.map(function (g, i) {
    return '<button class="gallery-item reveal" type="button" data-src="' + esc(g.src) +
      '" data-cap="' + esc(g.title || "") + (g.desc ? " · " + esc(g.desc) : "") +
      '" aria-label="查看大图：' + esc(g.title || ("作品 " + (i + 1))) + '">' +
      '<img src="' + esc(g.src) + '" alt="' + esc(g.title || ("作品 " + (i + 1))) + '" loading="lazy" decoding="async" />' +
      '<span class="gallery-zoom" aria-hidden="true">' + ICONS.zoom + '</span>' +
      '<span class="gallery-cap">' + esc(g.title || ("作品 " + (i + 1))) +
        (g.desc ? '<em>' + esc(g.desc) + '</em>' : '') + '</span>' +
    '</button>';
  }).join("");

  /* 图片加载失败时自动隐藏该卡片；全部失败则隐藏整个模块 */
  $$(".gallery-item img", grid).forEach(function (img) {
    img.addEventListener("error", function () {
      const item = img.closest(".gallery-item");
      if (item) item.style.display = "none";
      if (!$$(".gallery-item", grid).some(function (el) { return el.style.display !== "none"; })) {
        const sec = $("#gallery");
        if (sec) sec.style.display = "none";
      }
    });
  });
}

/* =========================================================
   4. Tabs（ARIA + 键盘）
   ========================================================= */
function initTabs(container) {
  const tabs = $$('[role="tab"]', container);
  const panels = $$('[role="tabpanel"]', container);
  const list = $('[role="tablist"]', container);
  if (!tabs.length || !list) return;

  function activate(idx, focus) {
    tabs.forEach(function (t, i) {
      const on = i === idx;
      t.setAttribute("aria-selected", String(on));
      t.tabIndex = on ? 0 : -1;
      if (panels[i]) panels[i].hidden = !on;
    });
    const panel = panels[idx];
    if (panel) {
      $$(".reveal", panel).forEach(function (el) { el.classList.add("in"); });
      $$(".bar", panel).forEach(function (b) {
        const inner = b.firstElementChild;
        if (inner) inner.style.width = b.dataset.w + "%";
      });
    }
    if (focus) tabs[idx].focus();
  }

  tabs.forEach(function (t, i) {
    t.addEventListener("click", function () { activate(i); });
  });

  list.addEventListener("keydown", function (e) {
    const i = tabs.indexOf(document.activeElement);
    if (i < 0) return;
    let n = null;
    if (e.key === "ArrowRight") n = (i + 1) % tabs.length;
    else if (e.key === "ArrowLeft") n = (i - 1 + tabs.length) % tabs.length;
    else if (e.key === "Home") n = 0;
    else if (e.key === "End") n = tabs.length - 1;
    if (n !== null) {
      e.preventDefault();
      activate(n, true);
    }
  });

  activate(0);
}

function setupLightbox() {
  const lb = $("#lightbox");
  const lbImg = $("#lbImg");
  const lbCap = $("#lbCap");
  const lbClose = $("#lbClose");
  if (!lb || !lbImg) return;
  let lastFocus = null;

  function open(src, cap) {
    lastFocus = document.activeElement;
    lbImg.src = src;
    lbImg.alt = cap || "";
    lbCap.textContent = cap || "";
    lb.hidden = false;
    requestAnimationFrame(function () { lb.classList.add("show"); });
    document.body.style.overflow = "hidden";
    lbClose.focus();
  }

  function close() {
    lb.classList.remove("show");
    document.body.style.overflow = "";
    setTimeout(function () {
      lb.hidden = true;
      lbImg.src = "";
    }, 230);
    if (lastFocus && lastFocus.focus) lastFocus.focus();
  }

  document.addEventListener("click", function (e) {
    const item = e.target.closest(".gallery-item");
    if (item) {
      open(item.dataset.src, item.dataset.cap);
      return;
    }
    if (e.target === lb) close();
  });

  lbClose.addEventListener("click", close);

  document.addEventListener("keydown", function (e) {
    if (e.key === "Escape" && !lb.hidden) close();
  });
}

function setupTheme() {
  const btn = $("#themeBtn");
  const root = document.documentElement;

  function paint() {
    const dark = root.dataset.theme === "dark";
    btn.innerHTML = dark ? ICONS.sun : ICONS.moon;
    btn.setAttribute("aria-label", dark ? "切换到浅色模式" : "切换到深色模式");
  }

  btn.addEventListener("click", function () {
    root.dataset.theme = root.dataset.theme === "dark" ? "light" : "dark";
    try { localStorage.setItem("hgy-theme", root.dataset.theme); } catch (e) {}
    paint();
  });

  paint();
}

function toast(msg) {
  const t = $("#toast");
  t.textContent = msg;
  t.classList.add("show");
  clearTimeout(toast._timer);
  toast._timer = setTimeout(function () { t.classList.remove("show"); }, 2200);
}

function setupCopy() {
  document.addEventListener("click", function (e) {
    const btn = e.target.closest("#copyMail");
    if (!btn) return;
    const done = function () { toast("邮箱已复制：" + EMAIL); };
    if (navigator.clipboard && navigator.clipboard.writeText) {
      navigator.clipboard.writeText(EMAIL).then(done).catch(fallback);
    } else {
      fallback();
    }
    function fallback() {
      const ta = document.createElement("textarea");
      ta.value = EMAIL;
      ta.setAttribute("readonly", "");
      ta.style.position = "fixed";
      ta.style.left = "-9999px";
      document.body.appendChild(ta);
      ta.select();
      try { document.execCommand("copy"); done(); }
      catch (err) { toast("复制失败，请手动复制：" + EMAIL); }
      document.body.removeChild(ta);
    }
  });
}

function setupToTop() {

  btn.addEventListener("click", function () {

  });
}

function setupNavHighlight() {
  const links = $$(".nav-links a");
  if (!("IntersectionObserver" in window) || !links.length) return;
  const map = {};
  links.forEach(function (a) { map[a.getAttribute("href").slice(1)] = a; });
  const obs = new IntersectionObserver(function (entries) {
    entries.forEach(function (en) {
      if (!en.isIntersecting) return;
      const id = en.target.id;
      links.forEach(function (a) { a.classList.remove("active"); });
      if (map[id]) map[id].classList.add("active");
    });
  }, { rootMargin: "-45% 0px -50% 0px", threshold: 0 });
  $$("main section[id]").forEach(function (s) { obs.observe(s); });
}

function setupReveal() {
  const bars = $$(".bar");
  const reveals = $$(".reveal");

  if (!("IntersectionObserver" in window)) {
    reveals.forEach(function (el) { el.classList.add("in"); });
    bars.forEach(function (b) { b.firstElementChild.style.width = b.dataset.w + "%"; });
    return;
  }

  const io = new IntersectionObserver(function (entries) {
    entries.forEach(function (en) {
      if (!en.isIntersecting) return;
      const el = en.target;
      el.classList.add("in");
      if (el.classList.contains("bar") && el.firstElementChild) {
        el.firstElementChild.style.width = el.dataset.w + "%";
      }
      io.unobserve(el);
    });
  }, { threshold: 0.12, rootMargin: "0px 0px -6% 0px" });

  reveals.forEach(function (el) { io.observe(el); });
  bars.forEach(function (el) { io.observe(el); });
}
